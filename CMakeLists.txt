cmake_minimum_required(VERSION 3.16)
project(PS4DownloadManager VERSION 0.1.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_library(dm src/networking/http.cpp src/storage/storage.cpp src/downloads/engine.cpp src/realdebrid/debrid.cpp src/platform/curl.cpp)
target_include_directories(dm PUBLIC include)
target_compile_definitions(dm PUBLIC _FILE_OFFSET_BITS=64)
target_link_libraries(dm PUBLIC CURL::libcurl Threads::Threads)
add_executable(ps4dm-cli src/main.cpp)
target_link_libraries(ps4dm-cli PRIVATE dm)
enable_testing()
add_executable(dm-tests tests/test.cpp)
target_link_libraries(dm-tests PRIVATE dm)
add_test(NAME core COMMAND dm-tests)
add_executable(dm-controls tests/controls.cpp)
target_link_libraries(dm-controls PRIVATE dm)
